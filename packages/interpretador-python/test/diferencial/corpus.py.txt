def bolha(v):
    v = v[:]
    for i in range(len(v)):
        for j in range(len(v) - 1 - i):
            if v[j] > v[j + 1]: v[j], v[j + 1] = v[j + 1], v[j]
    return v
def busca(v, x):
    lo, hi = 0, len(v) - 1
    while lo <= hi:
        m = (lo + hi) // 2
        if v[m] == x: return m
        lo, hi = (m + 1, hi) if v[m] < x else (lo, m - 1)
    return -1
def base(n, b):
