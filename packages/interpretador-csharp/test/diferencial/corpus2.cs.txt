using System;
using System.Collections.Generic;
using System.Linq;
class Produto {
    public int Id { get; set; }
    public string Nome { get; set; }
    public decimal Preco { get; set; }
    public int Qtd { get; set; }
    public override string ToString() => $"{Id,3} | {Nome,-12} | {Preco,10:C2} | {Qtd,4}";
}
class Estoque {
    private readonly List<Produto> itens = new List<Produto>();
    private int proximo = 1;
    public Produto Adicionar(string nome, decimal preco, int qtd) { var p = new Produto { Id = proximo++, Nome = nome, Preco = preco, Qtd = qtd }; itens.Add(p); return p; }
    public bool Remover(int id) => itens.RemoveAll(p => p.Id == id) > 0;
    public decimal Total() => itens.Sum(p => p.Preco * p.Qtd);
    public IEnumerable<Produto> Baixo(int limite) => itens.Where(p => p.Qtd < limite).OrderBy(p => p.Qtd);
    public Produto this[int id] => itens.FirstOrDefault(p => p.Id == id);
    public IEnumerable<Produto> Todos => itens;
}
class Program {
    static void Main() {
        var e = new Estoque();
        e.Adicionar("Caneta", 1.5m, 100); e.Adicionar("Caderno", 12.9m, 5); e.Adicionar("Mochila", 149.9m, 2); e.Adicionar("Lápis", 0.8m, 300);
        foreach (var p in e.Todos) Console.WriteLine(p);
        Console.WriteLine($"Total: {e.Total():N2}");
        Console.WriteLine(string.Join(", ", e.Baixo(10).Select(p => p.Nome)));
        Console.WriteLine(e.Remover(2) + " " + e.Remover(99) + " " + (e[2] == null) + " " + e[3].Nome);
        decimal media = e.Todos.Average(p => p.Preco); Console.WriteLine(Math.Round(media, 2) + " " + media.ToString("F3"));
        Console.WriteLine(e.Todos.Max(p => p.Preco) + " " + e.Todos.OrderByDescending(p => p.Preco).First().Nome);
        Console.Write("Opcao: "); string op = Console.ReadLine();
        switch (op) { case "1": Console.WriteLine("listar"); break; case "2": Console.WriteLine("sair"); break; default: Console.WriteLine("invalida: " + op); break; }
        Console.Write("Qtd: "); if (int.TryParse(Console.ReadLine(), out int q) && q > 0) Console.WriteLine("ok " + q); else Console.WriteLine("qtd invalida");
        Console.Write("Preco: "); decimal pr; while (!decimal.TryParse(Console.ReadLine(), out pr)) Console.Write("de novo: "); Console.WriteLine($"{pr:F2}");
    }
}

using System;

interface IForma { double Area(); double Perimetro(); string Nome { get; } }
interface IDesenhavel { void Desenhar(); }
abstract class Base : IForma, IDesenhavel, IComparable<Base> {
    public abstract double Area();
    public abstract double Perimetro();
    public virtual string Nome => GetType().Name;
    public void Desenhar() => Console.WriteLine($"[{Nome}] area={Area():F2} perim={Perimetro():F2}");
    public int CompareTo(Base o) => Area().CompareTo(o.Area());
    public override bool Equals(object o) => o is Base b && Math.Abs(b.Area() - Area()) < 1e-9;
    public override int GetHashCode() => Area().GetHashCode();
}
class Quad : Base { double l; public Quad(double l) { this.l = l; } public override double Area() => l * l; public override double Perimetro() => 4 * l; }
class Circ : Base { double r; public Circ(double r) { this.r = r; } public override double Area() => Math.PI * r * r; public override double Perimetro() => 2 * Math.PI * r; public override string Nome => "Circulo"; }
class Tri : Base { double b, h; public Tri(double b, double h) { this.b = b; this.h = h; } public override double Area() => b * h / 2; public override double Perimetro() => b + h + Math.Sqrt(b * b + h * h); }
class Program {

        foreach (IDesenhavel d in fs.OfType<IDesenhavel>().Take(1)) d.Desenhar();

    }

using System;
using System.Collections.Generic;
class Fracao {
    public int N { get; }
    public int D { get; }
    public Fracao(int n, int d) { if (d == 0) throw new ArgumentException("denominador zero"); int g = Mdc(Math.Abs(n), Math.Abs(d)); if (d < 0) { n = -n; d = -d; } N = n / g; D = d / g; }
    static int Mdc(int a, int b) => b == 0 ? (a == 0 ? 1 : a) : Mdc(b, a % b);
    public static Fracao operator +(Fracao a, Fracao b) => new Fracao(a.N * b.D + b.N * a.D, a.D * b.D);
    public static Fracao operator *(Fracao a, Fracao b) => new Fracao(a.N * b.N, a.D * b.D);
    public static Fracao operator -(Fracao a) => new Fracao(-a.N, a.D);
    public static bool operator ==(Fracao a, Fracao b) => a.N == b.N && a.D == b.D;
    public static bool operator !=(Fracao a, Fracao b) => !(a == b);
    public static implicit operator double(Fracao f) => (double)f.N / f.D;
    public static explicit operator Fracao(int i) => new Fracao(i, 1);
    public override bool Equals(object o) => o is Fracao f && this == f;
    public override int GetHashCode() => N * 31 + D;
    public override string ToString() => D == 1 ? N.ToString() : $"{N}/{D}";
}
