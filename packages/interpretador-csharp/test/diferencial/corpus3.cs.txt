class P {

        int a = -16; long b = -16L; uint u = 0xF0000000; ulong ul = ulong.MaxValue;

using System;
class P {
    static void Main() {
        int x = 5; double pi = 3.14159; bool b = true; string nome = "Ana";
        Console.WriteLine($"""v={x} pi={pi:F2} [{x,4}] {(b ? "sim" : "nao")} "{nome}" """);
        Console.WriteLine($$"""{x}={{x}} e {{nome.ToUpper()}}""");
        string t = $"""
            Nome: {nome}
              Idade: {x + 20}
            Total: {pi * x:F1}
            """;
        Console.WriteLine(t);
        string j = $$"""
            {
              "nome": "{{nome}}",
              "x": {{x}},
              "lista": [{{x}}, {{x * 2}}]
            }
            """;
        Console.WriteLine(j);
        Console.WriteLine($$$"""a {{b}} {{{x}}} {{{{{nome}}}}}""");
        int neg = -64;
        Console.WriteLine($"""desl={neg >>> 28} {neg >> 28}""");
        Console.WriteLine("fim");
    }
}
