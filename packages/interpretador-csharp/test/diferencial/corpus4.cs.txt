using System;
using System.Collections.Generic;
class P {
    static string Descrever(int[] v) => v switch {
        [] => "vazio",
        [var unico] => "um: " + unico,
        [1, 2, ..] => "comeca com 1,2",
        [_, _] => "dois quaisquer",
        [var primeiro, .., var ultimo] when primeiro == ultimo => "pontas iguais " + primeiro,
        [var primeiro, .. var meio, var ultimo] => $"primeiro={primeiro} ultimo={ultimo} meio={meio.Length}",
    };
    static string Lista(List<int> l) => l switch {
        [0, ..] => "zero na frente",
        [.., 9] => "nove no fim",
        [var a, var b, var c] => $"tres: {a + b + c}",
        _ => "outra",
    };
    static void Main() {
        int[] a = { 1, 2, 3, 4 };
        Console.WriteLine(a is [1, 2, ..]);
        Console.WriteLine(a is [1, 2]);
        Console.WriteLine(a is [_, _, _, _]);
        Console.WriteLine(a is [var f, .., var l] && f < l);
        Console.WriteLine(a is [1, .. var resto] ? resto.Length + ":" + string.Join(",", resto) : "nao");
        Console.WriteLine(a is [.., 4]);
        Console.WriteLine(a is [.., > 5]);
        Console.WriteLine(a is not [1, ..]);
        foreach (var v in new[] { new int[0], new[] { 7 }, new[] { 1, 2, 3 }, new[] { 5, 6 }, new[] { 4, 0, 4 }, new[] { 3, 1, 2, 8 } })
            Console.WriteLine(Descrever(v));
        Console.WriteLine(Lista(new List<int> { 0, 5 }));
        Console.WriteLine(Lista(new List<int> { 1, 9 }));
        Console.WriteLine(Lista(new List<int> { 1, 2, 3 }));
        Console.WriteLine(Lista(new List<int>()));
        string s = "abc";
        Console.WriteLine(s is ['a', ..]);
        Console.WriteLine(s is [.., 'c']);
        Console.WriteLine(s is [var x, var y, var z] ? "" + z + y + x : "?");
        Console.WriteLine(s is ['a', .. var cauda] ? cauda : "-");
        Console.WriteLine("" is []);
        int[] nulo = null;
        Console.WriteLine(nulo is [..]);
        switch (a) {
            case [1, _, ..]: Console.WriteLine("instrucao: comeca com 1"); break;
            default: Console.WriteLine("instrucao: outra"); break;
        }
        string[] nomes = { "ana", "bia", "caio" };
        if (nomes is [var p, _, var q]) Console.WriteLine(p + "&" + q);
    }
}
