#include <iostream>
#include <set>
#include <map>
#include <queue>
#include <vector>
#include <string>
#include <algorithm>
using namespace std;
struct Ponto {
    int x, y;
    bool operator<(const Ponto &o) const { return x != o.x ? x < o.x : y < o.y; }
};
struct PorTamanho {
    bool operator()(const string &a, const string &b) const { return a.size() != b.size() ? a.size() < b.size() : a < b; }
};
struct Tarefa { string nome; int prioridade; };
struct MenosUrgente {
    bool operator()(const Tarefa &a, const Tarefa &b) const { return a.prioridade < b.prioridade; }
};
int main() {
    set<Ponto> pontos = {{3, 1}, {1, 2}, {1, 1}, {2, 5}, {1, 2}};
    for (auto &p : pontos) cout << "(" << p.x << "," << p.y << ")";
    cout << endl;
    set<string, PorTamanho> s = {"banana", "kiwi", "uva", "abacaxi", "figo"};
    for (auto &w : s) cout << w << " ";
    cout << endl;
    map<string, int, greater<string>> m = {{"a", 1}, {"c", 3}, {"b", 2}};
    for (auto &[k, v] : m) cout << k << v;
    cout << endl;
    priority_queue<Tarefa, vector<Tarefa>, MenosUrgente> fila;
    fila.push({"lavar", 2});
    fila.push({"estudar", 5});
    fila.push({"comer", 3});
    fila.push({"dormir", 5});
    while (!fila.empty()) { cout << fila.top().nome << " "; fila.pop(); }
    cout << endl;
    auto cmp = [](int a, int b) { return a % 10 < b % 10; };
    vector<int> v = {23, 41, 12, 55, 30, 9, 61};
    stable_sort(v.begin(), v.end(), cmp);
    for (int x : v) cout << x << ' ';
    cout << endl;
    priority_queue<int, vector<int>, greater<int>> menor;
    for (int x : {5, 1, 8, 3, 9, 2}) menor.push(x);
    cout << menor.top() << " " << menor.size() << endl;
    return 0;
}
