#include <iostream>
#include <vector>
#include <queue>
#include <climits>
using namespace std;
int main() {
    int n = 5;
    vector<vector<pair<int, int>>> adj(n);
    auto aresta = [&](int a, int b, int w) { adj[a].push_back({b, w}); adj[b].push_back({a, w}); };
    aresta(0, 1, 4); aresta(0, 2, 1); aresta(2, 1, 2); aresta(1, 3, 1); aresta(2, 3, 5); aresta(3, 4, 3);
    vector<int> dist(n, INT_MAX);
    priority_queue<pair<int, int>, vector<pair<int, int>>, greater<pair<int, int>>> pq;
    dist[0] = 0;
    pq.push({0, 0});
    while (!pq.empty()) {
        auto [d, u] = pq.top();
        pq.pop();
        if (d > dist[u]) continue;
        for (auto &[v, w] : adj[u]) {
            if (dist[u] + w < dist[v]) {
                dist[v] = dist[u] + w;
                pq.push({dist[v], v});
            }
        }
    }
    for (int i = 0; i < n; i++) cout << "dist[" << i << "] = " << dist[i] << endl;
    priority_queue<int> maxima;
    for (int x : {3, 8, 1, 9, 4}) maxima.push(x);
    while (!maxima.empty()) { cout << maxima.top() << ' '; maxima.pop(); }
    cout << endl;
    return 0;
}

using namespace std;
bool balanceado(const string &s) {
    stack<char> p;
    for (char c : s) {
        if (c == '(' || c == '[') p.push(c);
        else if (c == ')' || c == ']') {
            if (p.empty()) return false;
            char t = p.top(); p.pop();
            if ((c == ')' && t != '(') || (c == ']' && t != '[')) return false;
        }
    }
    return p.empty();
}
int main() {
    cout << balanceado("([()])") << balanceado("([)]") << balanceado("((") << endl;
    queue<int> f;

#include <iostream>
#include <vector>
#include <string>
using namespace std;
class Conta {
    string titular;
    double saldo;
public:
    Conta(string t, double s) : titular(t), saldo(s) {}
    void depositar(double v) { saldo += v; }
    bool sacar(double v) {
        if (v > saldo) return false;
        saldo -= v;
        return true;
    }
    double getSaldo() const { return saldo; }
    string getTitular() const { return titular; }
};
int main() {
    vector<Conta> contas;
    contas.emplace_back("Ana", 100);
    contas.push_back(Conta("Beto", 50));
    contas[0].depositar(25);
    cout << contas[1].sacar(80) << contas[1].sacar(30) << endl;
    double total = 0;
    for (const Conta &c : contas) {
        cout << c.getTitular() << ": " << c.getSaldo() << endl;
        total += c.getSaldo();
    }
    cout << "total " << total << endl;
    vector<vector<int>> m(3, vector<int>(3, 0));
    for (int i = 0; i < 3; i++) for (int j = 0; j < 3; j++) m[i][j] = i * 3 + j;
    for (auto &linha : m) {
        for (int x : linha) cout << x << ' ';
        cout << endl;
    }
    return 0;
}
