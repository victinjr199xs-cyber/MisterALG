#include <bits/stdc++.h>
using namespace std;
int main() {
    int n = 5;
    vector<vector<pair<int,int>>> g(n);
    g[0].push_back({1, 4}); g[0].push_back({2, 1}); g[2].push_back({1, 2});
    g[1].push_back({3, 1}); g[2].push_back({3, 5}); g[3].push_back({4, 3});
    vector<int> d(n, INT_MAX);
    priority_queue<pair<int,int>, vector<pair<int,int>>, greater<pair<int,int>>> pq;
    d[0] = 0; pq.push({0, 0});
    while (!pq.empty()) {
        auto [du, u] = pq.top(); pq.pop();
        if (du > d[u]) continue;
        for (auto [v, w] : g[u]) if (d[u] + w < d[v]) { d[v] = d[u] + w; pq.push({d[v], v}); }
    }
    for (int i = 0; i < n; i++) cout << i << ":" << d[i] << "\n";
}
