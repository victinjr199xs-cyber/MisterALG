}

int fib(int n) { return n < 2 ? n : fib(n - 1) + fib(n - 2); }
void hanoi(int n, char a, char b, char c, int *cont) {
    if (n == 0) return;
    hanoi(n - 1, a, c, b, cont);
    (*cont)++;
    if (*cont <= 5) printf("mover %d de %c para %c\n", n, a, c);
    hanoi(n - 1, c, b, a, cont);
}

int pot(int b, int e) { if (e == 0) return 1; int m = pot(b, e / 2); return e % 2 ? m * m * b : m * m; }
