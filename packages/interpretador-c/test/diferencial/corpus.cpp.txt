#include <iostream>
#include <string>
using namespace std;
class Animal {
protected: string nome;
public:
    Animal(string n) : nome(n) {}
    virtual ~Animal() { cout << "~Animal " << nome << endl; }
    virtual string som() const { return "..."; }
    void fala() const { cout << nome << " diz " << som() << endl; }
};
class Cao : public Animal {
public:
    Cao(string n) : Animal(n) {}
    string som() const override { return "Au"; }
    ~Cao() { cout << "~Cao" << endl; }
};
class Gato : public Animal { public: Gato(string n) : Animal(n) {} string som() const override { return "Miau"; } };
struct Conta {
    double saldo = 0; static int total;
    Conta() { total++; }
    void depositar(double v) { saldo += v; }
    bool sacar(double v) { if (v > saldo) return false; saldo -= v; return true; }
};
int Conta::total = 0;
int main() {
    Animal *a[3] = {new Cao("Rex"), new Gato("Tom"), new Animal("Bicho")};
    for (int i = 0; i < 3; i++) a[i]->fala();
    for (int i = 0; i < 3; i++) delete a[i];
    Conta c, d; c.depositar(100.5); cout << c.sacar(50) << c.sacar(100) << " " << c.saldo << " " << Conta::total << endl;
    return 0;
}

#include <iostream>
using namespace std;
template <typename T> T maior(T a, T b) { return a > b ? a : b; }
template <typename T> class Pilha {
    T dados[10]; int topo = 0;
public:
    void push(T x) { dados[topo++] = x; }
    T pop() { return dados[--topo]; }
    bool vazia() const { return topo == 0; }
};
void troca(int &a, int &b) { int t = a; a = b; b = t; }
int &ref(int *v, int i) { return v[i]; }
int main() {
    cout << maior(3, 7) << " " << maior(2.5, 1.5) << " " << maior('a', 'z') << endl;
    Pilha<int> p; for (int i = 1; i <= 4; i++) p.push(i * i);
    while (!p.vazia()) cout << p.pop() << " "; cout << endl;
    int a = 1, b = 2; troca(a, b); cout << a << b << endl;
    int v[3] = {1, 2, 3}; ref(v, 1) = 20; cout << v[1] << endl;
    const int &r = a; a = 99; cout << r << endl;
    auto x = 5 / 2.0; cout << x << " " << sizeof(x) << endl;
    return 0;
}

#include <iostream>
using namespace std;
struct V2 {
    double x, y;
    V2(double x = 0, double y = 0) : x(x), y(y) {}
    V2 operator+(const V2 &o) const { return V2(x + o.x, y + o.y); }
    V2 operator*(double k) const { return V2(x * k, y * k); }
    bool operator==(const V2 &o) const { return x == o.x && y == o.y; }
    V2 &operator+=(const V2 &o) { x += o.x; y += o.y; return *this; }
};
ostream &operator<<(ostream &os, const V2 &v) { return os << "(" << v.x << ", " << v.y << ")"; }
int main() {
    V2 a(1, 2), b(3.5, -1);
    cout << a + b << " " << a * 2 << " " << (a == b) << endl;
    a += b; cout << a << endl;
    int i = 5; cout << i++ << ++i << i-- << --i << endl;
    cout << (5 > 3 ? "sim" : "nao") << endl;
    unsigned u = 0; u--; cout << u << endl;
    cout << (char)('a' + 2) << (int)'a' << endl;
    return 0;
}
