create table dept (id int primary key, nome text);
create table func (id int primary key, nome text, dept_id int references dept(id), salario numeric(10,2));
insert into dept values (1, 'TI'), (2, 'RH'), (3, 'Vazio');
insert into func values (1, 'Ana', 1, 5000), (2, 'Bruno', 1, 4500.5), (3, 'Carla', 2, 3000), (4, 'Dani', null, 2000);
select f.nome, d.nome as dept, f.salario from func f join dept d on d.id = f.dept_id order by f.id;
select f.nome, d.nome as dept from func f left join dept d on d.id = f.dept_id order by f.id;
select d.nome, count(f.id) as qtd, sum(f.salario) as total, avg(f.salario)::numeric(10,2) as media, max(f.salario) as maior from dept d left join func f on f.dept_id = d.id group by d.nome order by d.nome;
select d.nome, string_agg(f.nome, ', ' order by f.nome) as nomes from dept d join func f on f.dept_id = d.id group by d.nome having count(*) > 0 order by 1;
select nome, salario, rank() over (order by salario desc) as r, sum(salario) over () as soma, salario / sum(salario) over () * 100 as pct from func order by salario desc;
with ranked as (select *, row_number() over (partition by dept_id order by salario desc) rn from func) select nome, dept_id, rn from ranked where rn = 1 order by nome;
select * from func where dept_id in (select id from dept where nome = 'TI') order by id;
select case when salario > 4000 then 'alto' when salario > 2500 then 'medio' else 'baixo' end as faixa, count(*) from func group by 1 order by 1;
select * from func cross join dept where func.id < 2 order by dept.id;
